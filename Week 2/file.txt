create table users(

	userId serial primary key not null,
	userName varchar(200) not null,
	city varchar(200) not null,
	userDetials text not null,
	created_at timestamptz default now()
)

-- alter table users add primary key(userId);

-- select * from users;

create table post(
	postId serial primary key not null,
	tittle varchar(200) not null,
	contents text not null,
	userId serial not null,
	created_at timestamptz default now() not null,
	constraint fk_post_user foreign key (userId) references users(userId) on delete cascade
)

alter table post alter column userId type int

create table command(

	commandId serial primary key not null,
	command text,
	creatd_at timestamptz default now() not null,

	postId serial not null,
	userId serial not null,

	constraint fk_command_user foreign key (userId) references users(userId) on delete cascade,
	constraint fk_command_post foreign key (postId) references post(postId) on delete cascade
)
alter table command alter column userId type int
alter table command alter column postId type int
